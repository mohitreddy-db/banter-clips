# Note on the signed-in app routes (/studio, /clips, /account, /pricing,
# /onboarding): they are deliberately NOT disallowed here. They serve
# <meta name="robots" content="noindex"> instead — see src/lib/seo.js.
# A Disallow rule would stop Google fetching those pages, and a noindex it
# never fetches is a noindex it never honours, so anything already indexed
# would be stranded. Once Search Console reports them dropped from the index,
# they can be moved to Disallow lines to save crawl budget.

User-agent: *
Allow: /

# Admin surface — no crawl, ever.
Disallow: /admin

# Build output and the auth callback carry no content worth indexing.
Disallow: /assets/
Disallow: /reset-password

Sitemap: https://www.banterclips.com/sitemap.xml
